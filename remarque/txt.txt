def get_liste_configuration(cur):
    cur.execute("SELECT * FROM v_configuration_courante WHERE id_statut =! 14")
    return cur.fetchall()

def get_liste_archiver(cur):
    cur.execute("SELECT * FROM v_configuration_courante WHERE id_statut = 14")
    return cur.fetchall()

def get_configuration_by_id(cur, id):
    cur.execute("SELECT * FROM t_configuration WHERE id = %s", (id,))
    return cur.fetchone()

def get_historique_by_type(cur, type):
    cur.execute("SELECT * FROM t_configuration WHERE type = %s ORDER BY id DESC", (type,))
    return cur.fetchall()

def get_valeur_configuration(cur, type):
    cur.execute("SELECT valeur FROM v_configuration_courante WHERE type = %s AND id_statut <> 14", (type,))
    return cur.fetchone()

def get_dernier_statut(cur, id_configuration):
    cur.execute(
        "SELECT id_statut FROM t_configuration_statut WHERE id_configuration = %s ORDER BY id DESC LIMIT 1",
        (id_configuration,)
    )
    result = cur.fetchone()
    return result[0] if result else None
